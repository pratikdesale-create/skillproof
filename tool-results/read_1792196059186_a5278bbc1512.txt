   456→function InterviewPanel({
   457→  jobId, existingApp, appsLoading,
   458→}: {
   459→  jobId: string
   460→  existingApp: ApplicationRow | null
   461→  appsLoading: boolean
   462→}) {
   463→  const navigate = useAppStore(s => s.navigate)
   464→  const qc = useQueryClient()
   465→  const [open, setOpen] = useState(false)
   466→  const [scheduledAt, setScheduledAt] = useState('')
   467→  const [duration, setDuration] = useState('')
   468→  const [location, setLocation] = useState('')
   469→  const [meetingUrl, setMeetingUrl] = useState('')
   470→  const [type, setType] = useState<string>('TECHNICAL')
   471→  const [round, setRound] = useState('1')
   472→  const [notes, setNotes] = useState('')
   473→
   474→  const mutation = useMutation({
   475→    mutationFn: async () => {
   476→      if (!existingApp) throw new Error('No existing application for this candidate')
   477→      const body: Record<string, unknown> = {
   478→        scheduledAt: new Date(scheduledAt).toISOString(),
   479→        type,
   480→        round: parseInt(round, 10) || 1,
   481→      }
   482→      if (duration.trim()) body.duration = parseInt(duration, 10)
   483→      if (location.trim()) body.location = location.trim()
   484→      if (meetingUrl.trim()) body.meetingUrl = meetingUrl.trim()
   485→      if (notes.trim()) body.notes = notes.trim()
   486→      return apiPost(`/api/applications/${existingApp.id}/interview`, body)
   487→    },
   488→    onSuccess: () => {
   489→      toast.success('Interview scheduled')
   490→      qc.invalidateQueries({ queryKey: ['job-applications-count', jobId] })
   491→      qc.invalidateQueries({ queryKey: ['application', existingApp?.id] })
   492→      setOpen(false)
   493→      // Reset
   494→      setScheduledAt('')
   495→      setDuration('')
   496→      setLocation('')
   497→      setMeetingUrl('')
   498→      setNotes('')
   499→      setRound('1')
   500→      setType('TECHNICAL')
   501→      // Navigate to the application detail
   502→      if (existingApp) navigate(`recruiter/applications/${existingApp.id}`)
   503→    },
   504→    onError: (err: unknown) => {
   505→      const e = err as { message?: string }
   506→      toast.error(e.message ?? 'Failed to schedule interview')
   507→    },
   508→  })
   509→
   510→  if (appsLoading) {
   511→    return (
   512→      <Card>
   513→        <CardContent className="p-6 space-y-3">
   514→          <Skeleton className="h-4 w-32" />
   515→          <Skeleton className="h-12 w-full" />
   516→          <Skeleton className="h-12 w-full" />
   517→        </CardContent>
   518→      </Card>
   519→    )
   520→  }
   521→
   522→  if (!existingApp) {
   523→    return (
   524→      <Card className="border-amber-200 dark:border-amber-900 bg-amber-50/60 dark:bg-amber-950/40">
   525→        <CardHeader>
   526→          <CardTitle className="text-base flex items-center gap-2 text-amber-800 dark:text-amber-300">
   527→            <AlertTriangle className="size-4" /> No active application
   528→          </CardTitle>
   529→        </CardHeader>
   530→        <CardContent className="space-y-3">
   531→          <p className="text-sm text-amber-800 dark:text-amber-300">
