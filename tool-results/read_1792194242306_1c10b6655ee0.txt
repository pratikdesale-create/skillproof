   341→function UpdateStatusDialog({
   342→  open, onOpenChange, applicationId, currentStatus, allowedNext,
   343→}: {
   344→  open: boolean
   345→  onOpenChange: (open: boolean) => void
   346→  applicationId: string
   347→  currentStatus: ApplicationStatus
   348→  allowedNext: ApplicationStatus[]
   349→}) {
   350→  const qc = useQueryClient()
   351→  const [newStatus, setNewStatus] = useState<ApplicationStatus | ''>('')
   352→  const [reason, setReason] = useState('')
   353→
   354→  const mutation = useMutation({
   355→    mutationFn: (body: { status: ApplicationStatus; reason?: string }) =>
   356→      apiPatch(`/api/applications/${applicationId}`, body),
   357→    onSuccess: () => {
   358→      toast.success('Status updated')
   359→      onOpenChange(false)
   360→      setNewStatus('')
   361→      setReason('')
   362→      qc.invalidateQueries({ queryKey: ['application', applicationId] })
   363→      qc.invalidateQueries({ queryKey: ['officer-applications'] })
   364→      qc.invalidateQueries({ queryKey: ['officer-dashboard'] })
   365→    },
   366→    onError: (err: unknown) => {
   367→      const e = err as { message?: string }
   368→      toast.error(e.message ?? 'Failed to update status')
   369→    },
   370→  })
   371→
   372→  function submit() {
   373→    if (!newStatus) {
   374→      toast.error('Select a status')
   375→      return
   376→    }
   377→    mutation.mutate({ status: newStatus, reason: reason.trim() || undefined })
   378→  }
   379→
   380→  return (
   381→    <Dialog open={open} onOpenChange={(o) => { if (!o) { setNewStatus(''); setReason('') }; onOpenChange(o) }}>
   382→      <DialogContent>
   383→        <DialogHeader>
   384→          <DialogTitle>Update application status</DialogTitle>
   385→          <DialogDescription>
   386→            Current: <StatusBadge status={currentStatus} />
   387→            {allowedNext.length === 0 ? ' — terminal status, no further transitions.' : ' Select the next status.'}
   388→          </DialogDescription>
   389→        </DialogHeader>
   390→        <div className="space-y-3">
   391→          <div>
   392→            <Label className="text-xs text-slate-500">New status</Label>
   393→            <Select value={newStatus} onValueChange={(v) => setNewStatus(v as ApplicationStatus)}>
   394→              <SelectTrigger className="w-full"><SelectValue placeholder="Select status" /></SelectTrigger>
   395→              <SelectContent>
   396→                {allowedNext.map((s) => (
   397→                  <SelectItem key={s} value={s}>{s.replace(/_/g, ' ')}</SelectItem>
   398→                ))}
   399→              </SelectContent>
   400→            </Select>
   401→          </div>
   402→          <div>
   403→            <Label className="text-xs text-slate-500">Reason (optional)</Label>
   404→            <Textarea
   405→              rows={3}
   406→              value={reason}
   407→              onChange={(e) => setReason(e.target.value)}
   408→              placeholder="Explain the transition…"
   409→            />
   410→          </div>
   411→        </div>
   412→        <DialogFooter>
   413→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   414→          <Button onClick={submit} disabled={mutation.isPending || allowedNext.length === 0}>
   415→            {mutation.isPending ? <Loader2 className="size-4 mr-1 animate-spin" /> : null}
   416→            Update status
   417→          </Button>
   418→        </DialogFooter>
   419→      </DialogContent>
   420→    </Dialog>
   421→  )
   422→}

   424→function ScheduleInterviewDialog({
   425→  open, onOpenChange, applicationId,
   426→}: {
   427→  open: boolean
   428→  onOpenChange: (open: boolean) => void
   429→  applicationId: string
   430→}) {
   431→  const qc = useQueryClient()
   432→  const [scheduledAt, setScheduledAt] = useState('')
   433→  const [duration, setDuration] = useState('')
   434→  const [location, setLocation] = useState('')
   435→  const [meetingUrl, setMeetingUrl] = useState('')
   436→  const [type, setType] = useState<'TECHNICAL' | 'HR' | 'MANAGERIAL' | 'GROUP'>('TECHNICAL')
   437→  const [round, setRound] = useState('1')
   438→  const [notes, setNotes] = useState('')
   439→
   440→  function reset() {
   441→    setScheduledAt(''); setDuration(''); setLocation(''); setMeetingUrl('')
   442→    setType('TECHNICAL'); setRound('1'); setNotes('')
   443→  }
   444→
   445→  const mutation = useMutation({
   446→    mutationFn: (body: Record<string, unknown>) => apiPost(`/api/applications/${applicationId}/interview`, body),
   447→    onSuccess: () => {
   448→      toast.success('Interview scheduled — student notified')
   449→      onOpenChange(false)
   450→      reset()
   451→      qc.invalidateQueries({ queryKey: ['application', applicationId] })
   452→      qc.invalidateQueries({ queryKey: ['officer-applications'] })
   453→      qc.invalidateQueries({ queryKey: ['officer-dashboard'] })
   454→    },
   455→    onError: (err: unknown) => {
   456→      const e = err as { message?: string }
   457→      toast.error(e.message ?? 'Failed to schedule interview')
   458→    },
   459→  })
   460→
   461→  function submit() {
   462→    if (!scheduledAt) {
   463→      toast.error('Pick a date & time')
   464→      return
   465→    }
   466→    const body: Record<string, unknown> = {
   467→      scheduledAt: new Date(scheduledAt).toISOString(),
   468→      type,
   469→      round: Number(round) || 1,
   470→    }
   471→    if (duration) body.duration = Number(duration)
   472→    if (location) body.location = location
   473→    if (meetingUrl) body.meetingUrl = meetingUrl
   474→    if (notes) body.notes = notes
   475→    mutation.mutate(body)
   476→  }
   477→
   478→  // Default to next day 10am
   479→  function setDefaultTime() {
   480→    const d = new Date()
   481→    d.setDate(d.getDate() + 1)
   482→    d.setHours(10, 0, 0, 0)
   483→    const tz = d.getTimezoneOffset() * 60000
   484→    const local = new Date(d.getTime() - tz).toISOString().slice(0, 16)
   485→    setScheduledAt(local)
   486→  }
   487→
   488→  return (
   489→    <Dialog open={open} onOpenChange={(o) => { if (!o) reset(); onOpenChange(o) }}>
   490→      <DialogContent className="sm:max-w-lg">
   491→        <DialogHeader>
   492→          <DialogTitle>Schedule interview</DialogTitle>
   493→          <DialogDescription>If the application is at an earlier stage, it will be advanced to INTERVIEW.</DialogDescription>
   494→        </DialogHeader>
   495→        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   496→          <div className="space-y-1 md:col-span-2">
   497→            <Label className="text-xs text-slate-500">Scheduled at *</Label>
   498→            <div className="flex gap-2">
   499→              <Input
   500→                type="datetime-local"
   501→                value={scheduledAt}
   502→                onChange={(e) => setScheduledAt(e.target.value)}
   503→              />
   504→              <Button variant="outline" size="sm" onClick={setDefaultTime}>Now+1d 10am</Button>
   505→            </div>
   506→          </div>
   507→          <div className="space-y-1">
   508→            <Label className="text-xs text-slate-500">Type</Label>
   509→            <Select value={type} onValueChange={(v) => setType(v as typeof type)}>
   510→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   511→              <SelectContent>
   512→                {INTERVIEW_TYPES.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   513→              </SelectContent>
   514→            </Select>
   515→          </div>
   516→          <div className="space-y-1">
   517→            <Label className="text-xs text-slate-500">Round</Label>
   518→            <Input type="number" min={1} max={20} value={round} onChange={(e) => setRound(e.target.value)} />
   519→          </div>
   520→          <div className="space-y-1">
   521→            <Label className="text-xs text-slate-500">Duration (min)</Label>

   549→function ExtendOfferDialog({
   550→  open, onOpenChange, applicationId, existingOffer,
   551→}: {
   552→  open: boolean
   553→  onOpenChange: (open: boolean) => void
   554→  applicationId: string
   555→  existingOffer?: Offer | null
   556→}) {
   557→  const qc = useQueryClient()
   558→  const [amount, setAmount] = useState(existingOffer?.amount != null ? String(existingOffer.amount) : '')
   559→  const [currency, setCurrency] = useState(existingOffer?.currency ?? 'INR')
   560→  const [decision, setDecision] = useState<'PENDING' | 'ACCEPTED' | 'DECLINED'>(existingOffer?.decision as 'PENDING' | 'ACCEPTED' | 'DECLINED' ?? 'PENDING')
   561→  const [expiresAt, setExpiresAt] = useState('')
   562→
   563→  const mutation = useMutation({
   564→    mutationFn: (body: Record<string, unknown>) => apiPost(`/api/applications/${applicationId}/offer`, body),
   565→    onSuccess: () => {
   566→      toast.success('Offer updated — student notified')
   567→      onOpenChange(false)
   568→      qc.invalidateQueries({ queryKey: ['application', applicationId] })
   569→      qc.invalidateQueries({ queryKey: ['officer-applications'] })
   570→      qc.invalidateQueries({ queryKey: ['officer-dashboard'] })
   571→    },
   572→    onError: (err: unknown) => {
   573→      const e = err as { message?: string }
   574→      toast.error(e.message ?? 'Failed to extend offer')
   575→    },
   576→  })
   577→
   578→  function submit() {
   579→    const body: Record<string, unknown> = {
   580→      currency,
   581→      decision,
   582→    }
   583→    if (amount) body.amount = Number(amount)
   584→    if (expiresAt) body.expiresAt = new Date(expiresAt).toISOString()
   585→    mutation.mutate(body)
   586→  }
   587→
   588→  return (
   589→    <Dialog open={open} onOpenChange={onOpenChange}>
   590→      <DialogContent>
   591→        <DialogHeader>
   592→          <DialogTitle>
   593→            {existingOffer ? 'Update offer' : 'Extend offer'}
   594→          </DialogTitle>
   595→          <DialogDescription>
   596→            The application status will be moved to OFFER if it's at an earlier stage.
   597→          </DialogDescription>
   598→        </DialogHeader>
   599→        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   600→          <div className="space-y-1">
   601→            <Label className="text-xs text-slate-500">Amount</Label>
   602→            <Input type="number" min={0} value={amount} onChange={(e) => setAmount(e.target.value)} placeholder="e.g. 1200000" />
   603→          </div>
   604→          <div className="space-y-1">
   605→            <Label className="text-xs text-slate-500">Currency</Label>
   606→            <Input value={currency} onChange={(e) => setCurrency(e.target.value)} placeholder="INR" />
   607→          </div>
   608→          <div className="space-y-1">
   609→            <Label className="text-xs text-slate-500">Decision</Label>
   610→            <Select value={decision} onValueChange={(v) => setDecision(v as typeof decision)}>
   611→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   612→              <SelectContent>
   613→                {OFFER_DECISIONS.map((d) => <SelectItem key={d} value={d}>{d}</SelectItem>)}
   614→              </SelectContent>
   615→            </Select>
   616→          </div>
   617→          <div className="space-y-1">
   618→            <Label className="text-xs text-slate-500">Expires at</Label>
   619→            <Input type="date" value={expiresAt} onChange={(e) => setExpiresAt(e.target.value)} />
   620→          </div>
   621→        </div>
   622→        <DialogFooter>
   623→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   624→          <Button onClick={submit} disabled={mutation.isPending}>
   625→            {mutation.isPending ? <Loader2 className="size-4 mr-1 animate-spin" /> : null}
   626→            {existingOffer ? 'Update offer' : 'Extend offer'}
   627→          </Button>
   628→        </DialogFooter>
   629→      </DialogContent>
   630→    </Dialog>
   631→  )
   632→}
