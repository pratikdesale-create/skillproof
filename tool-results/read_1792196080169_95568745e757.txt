   412→function ResumePreview({ resume }: { resume: ResumeDetail }) {
   413→  const bullets = resume.bullets ?? []
   414→  if (bullets.length === 0) {
   415→    return (
   416→      <p className="text-sm text-slate-500 italic">This resume has no bullets.</p>
   417→    )
   418→  }
   419→  // Group bullets by section
   420→  const sections = new Map<string, ResumeBullet[]>()
   421→  for (const b of bullets) {
   422→    const arr = sections.get(b.section) ?? []
   423→    arr.push(b)
   424→    sections.set(b.section, arr)
   425→  }
   426→  return (
   427→    <div className="space-y-4">
   428→      {resume.isGenerated && (
   429→        <div className="text-xs text-slate-500 flex items-center gap-1">
   430→          <Badge variant="secondary" className="text-[10px]">Auto-generated</Badge>
   431→        </div>
   432→      )}
   433→      {resume.summary && (
   434→        <p className="text-sm text-slate-700 dark:text-slate-300 bg-slate-50 dark:bg-slate-900 rounded-md p-3">
   435→          {resume.summary}
   436→        </p>
   437→      )}
   438→      {Array.from(sections.entries()).map(([section, items]) => (
   439→        <div key={section}>
   440→          <h4 className="text-xs uppercase tracking-wider text-slate-500 mb-2">{section}</h4>
   441→          <ul className="space-y-1.5">
   442→            {items.map(b => (
   443→              <li key={b.id} className="text-sm flex items-start gap-2">
   444→                <span className="text-slate-300 dark:text-slate-600 mt-1">•</span>
   445→                <div className="flex-1">
   446→                  <span className="text-slate-700 dark:text-slate-300">{b.text}</span>
   447→                  {b.claimStatus && (
   448→                    <Badge
   449→                      variant="outline"
   450→                      className={cn(
   451→                        'ml-2 text-[10px]',
   452→                        b.claimStatus === 'SUPPORTED'
   453→                          ? 'bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-950 dark:text-emerald-300'
   454→                          : b.claimStatus === 'UNSUPPORTED'
   455→                          ? 'bg-rose-100 text-rose-700 border-rose-200 dark:bg-rose-950 dark:text-rose-300'
   456→                          : 'bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-950 dark:text-amber-300'
   457→                      )}
   458→                    >
   459→                      {b.claimStatus}
   460→                    </Badge>
   461→                  )}
   462→                </div>
   463→              </li>
   464→            ))}
   465→          </ul>
   466→        </div>
   467→      ))}
   468→    </div>
   469→  )
   470→}

   472→function UpdateStatusDialog({
   473→  open, onOpenChange, applicationId, currentStatus,
   474→}: {
   475→  open: boolean
   476→  onOpenChange: (open: boolean) => void
   477→  applicationId: string
   478→  currentStatus: ApplicationStatus
   479→}) {
   480→  const qc = useQueryClient()
   481→  const [newStatus, setNewStatus] = useState<string>(currentStatus)
   482→  const [reason, setReason] = useState('')
   483→
   484→  const mutation = useMutation({
   485→    mutationFn: () => apiPatch(`/api/applications/${applicationId}`, {
   486→      status: newStatus,
   487→      reason: reason.trim() || undefined,
   488→    }),
   489→    onSuccess: () => {
   490→      toast.success(`Application status updated to ${newStatus}`)
   491→      qc.invalidateQueries({ queryKey: ['application', applicationId] })
   492→      qc.invalidateQueries({ queryKey: ['recruiter-applications'] })
   493→      qc.invalidateQueries({ queryKey: ['recruiter-dashboard'] })
   494→      onOpenChange(false)
   495→      setReason('')
   496→    },
   497→    onError: (err: unknown) => {
   498→      const e = err as { message?: string }
   499→      toast.error(e.message ?? 'Failed to update status')
   500→    },
   501→  })
   502→
   503→  return (
   504→    <Dialog open={open} onOpenChange={onOpenChange}>
   505→      <DialogContent className="sm:max-w-lg">
   506→        <DialogHeader>
   507→          <DialogTitle>Update application status</DialogTitle>
   508→          <DialogDescription>
   509→            Current: <StatusBadge status={currentStatus} />. The API enforces allowed transitions; invalid ones will be rejected.
   510→          </DialogDescription>
   511→        </DialogHeader>
   512→        <div className="space-y-3">
   513→          <div className="space-y-1.5">
   514→            <Label>New status</Label>
   515→            <Select value={newStatus} onValueChange={setNewStatus}>
   516→              <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   517→              <SelectContent>
   518→                {ALL_STATUSES.map(s => <SelectItem key={s} value={s}>{s.replace(/_/g, ' ')}</SelectItem>)}
   519→              </SelectContent>
   520→            </Select>
   521→          </div>
   522→          <div className="space-y-1.5">
   523→            <Label htmlFor="status-reason">Reason (optional)</Label>
   524→            <Textarea
   525→              id="status-reason"
   526→              rows={3}
   527→              value={reason}
   528→              onChange={e => setReason(e.target.value)}
   529→              placeholder="e.g. Passed technical screen, advancing to HR round"
   530→            />
   531→          </div>

   690→function ExtendOfferDialog({

   794→function isoToDate(iso?: string | null): string {
   795→  if (!iso) return ''
   796→  const d = new Date(iso)
   797→  if (isNaN(d.getTime())) return ''
   798→  const pad = (n: number) => String(n).padStart(2, '0')
   799→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
   800→}
