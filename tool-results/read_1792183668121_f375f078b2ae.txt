   621→  for (const job of jobs) {
   622→    for (const student of students) {
   623→      const reqs = job.requirements.map(r => ({
   624→        skillId: r.skillId,
   625→        displayName: r.skill.displayName,
   626→        isRequired: r.isRequired,
   627→        weight: r.weight,
   628→      }))
   629→      const studentEvidence = job.requirements.map(r => {
   630→        const skillEvidence = student.evidence
   631→          .filter(e => e.skills.some(es => es.skillId === r.skillId))
   632→          .map(e => ({
   633→            evidenceId: e.id,
   634→            status: e.status,
   635→            relevance: e.skills.find(es => es.skillId === r.skillId)?.relevance ?? 1,
   636→            type: e.type,
   637→            title: e.title,
   638→          }))
   639→        return { skillId: r.skillId, evidence: skillEvidence }
   640→      })
   641→      const EVIDENCE_VALUE_RULES: Record<string, number> = {
   642→        VERIFIED: 1.0, PENDING_REVIEW: 0.25, UNVERIFIED: 0.25, REJECTED: 0, NEEDS_MORE_EVIDENCE: 0,
   643→      }
   644→      let totalWeightedMatch = 0, totalWeight = 0, reqCovered = 0, reqTotal = 0, prefCovered = 0, prefTotal = 0
   645→      const matchingSkills: any[] = []
   646→      const missingSkills: any[] = []
   647→      const breakdown: any[] = []
   648→      for (const req of reqs) {
   649→        const ev = studentEvidence.find(se => se.skillId === req.skillId)?.evidence ?? []
   650→        let max = 0, best: any = undefined
   651→        for (const e of ev) {
   652→          const base = EVIDENCE_VALUE_RULES[e.status] ?? 0
   653→          const value = base * Math.max(0, Math.min(1, e.relevance))
   654→          if (value > max) { max = value; best = e }
   655→        }
   656→        totalWeightedMatch += req.weight * max
   657→        totalWeight += req.weight
   658→        if (req.isRequired) {
   659→          reqTotal += 1
   660→          if (max >= 0.25) reqCovered += 1
   661→          if (max === 0) missingSkills.push({ skillId: req.skillId, name: req.displayName, weight: req.weight, isRequired: true, status: ev.length === 0 ? 'NO_EVIDENCE' : 'INSUFFICIENT_EVIDENCE' })
   662→        } else {
   663→          prefTotal += 1
   664→          if (max >= 0.25) prefCovered += 1
   665→          if (max === 0) missingSkills.push({ skillId: req.skillId, name: req.displayName, weight: req.weight, isRequired: false, status: ev.length === 0 ? 'NO_EVIDENCE' : 'INSUFFICIENT_EVIDENCE' })
   666→        }
   667→        if (max >= 0.25 && best) {
   668→          matchingSkills.push({ skillId: req.skillId, name: req.displayName, evidenceIds: ev.filter(e => EVIDENCE_VALUE_RULES[e.status] > 0).map(e => e.evidenceId) })
   669→        }
   670→        breakdown.push({ skillId: req.skillId, displayName: req.displayName, isRequired: req.isRequired, weight: req.weight, matchValue: max, bestEvidence: best, evidenceCount: ev.length })
   671→      }
   672→      const fitScore = totalWeight === 0 ? 0 : Math.round((totalWeightedMatch / totalWeight) * 1000) / 10
   673→      const requiredCoverage = reqTotal === 0 ? 1 : reqCovered / reqTotal
   674→      const preferredCoverage = prefTotal === 0 ? 1 : prefCovered / prefTotal
   675→
   676→      await db.matchResult.create({
   677→        data: {
   678→          jobId: job.id,
   679→          studentId: student.id,
   680→          fitScore,
   681→          requiredCoverage,
   682→          preferredCoverage,
   683→          matchingSkills: JSON.stringify(matchingSkills),
   684→          missingSkills: JSON.stringify(missingSkills),
   685→          explanation: JSON.stringify({ breakdown, factorsSummary: '', recommendation: '' }),
   686→          algorithmVersion: 'v1.0',
   687→          inputs: JSON.stringify({}),
   688→        },
   689→      })
   690→      matchCount++
   691→    }
   692→  }
